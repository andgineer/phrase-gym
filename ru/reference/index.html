class LocalBoundary:
    """Reject cross-site mutations and bound memory before JSON parsing."""

    def __init__(self, app, allowed_origins: tuple[str, ...]):
        self.app = app
        self.allowed_origins = allowed_origins

    def cross_origin(self, scope, headers) -> bool:
        if scope["method"] in ("GET", "HEAD", "OPTIONS"):
            return False
        origin = headers.get(b"origin", b"").decode()
        origin_url = urlsplit(origin)
        same_origin = (
            origin_url.netloc == headers.get(b"host", b"").decode()
            and origin_url.scheme == scope["scheme"]
        )
        cross_site = headers.get(b"sec-fetch-site") == b"cross-site"
        return cross_site or bool(
            origin and not same_origin and origin not in self.allowed_origins,
        )

    @staticmethod
    def too_large(headers) -> bool:
        try:
            return int(headers.get(b"content-length", b"0")) > MAX_REQUEST_BYTES
        except ValueError:
            return True

    async def __call__(self, scope, receive, send):
        if scope["type"] != "http":
            await self.app(scope, receive, send)
            return
        headers = dict(scope["headers"])
        if self.cross_origin(scope, headers):
            await JSONResponse({"detail": "Cross-origin mutations are not allowed"}, 403)(
                scope,
                receive,
                send,
            )
            return
        if self.too_large(headers):
            await JSONResponse({"detail": "Request exceeds 24 MiB"}, 413)(scope, receive, send)
            return
        body = bytearray()
        while True:
            message = await receive()
            if message["type"] == "http.disconnect":
                return
            body.extend(message.get("body", b""))
            if len(body) > MAX_REQUEST_BYTES:
                await JSONResponse({"detail": "Request exceeds 24 MiB"}, 413)(scope, receive, send)
                return
            if not message.get("more_body", False):
                break
        consumed = False

        async def replay():
            nonlocal consumed
            if not consumed:
                consumed = True
                return {"type": "http.request", "body": bytes(body), "more_body": False}
            return await receive()

        await self.app(scope, replay, send)

@click.command("export-existing")
@click.argument("namespace")
@click.argument("native_language", type=click.Choice(["en", "ru", "sr", "de"]))
@click.argument("output", type=click.Path(path_type=Path))
@click.pass_obj
def existing_export(config: Config, namespace: str, native_language: str, output: Path):
    """Prepare/regenerate existing target content without touching progress or recordings."""
    with runtime(config) as (jobs, _media), jobs.db.read() as transaction:
        items = transaction.scalars(
            select(SourceItem).where(SourceItem.namespace == namespace),
        ).all()
        phrases = []
        language = None
        target_sources = {}
        for item in items:
            target = transaction.get(Target, item.target_id)
            assert target is not None
            language = target.language
            revision = transaction.get(TargetRevision, (target.id, target.current_revision))
            assert revision is not None
            content = TargetInput.model_validate_json(revision.payload)
            target_sources[target.id] = item.source_id
            phrases.append(
                SourcePhrase(
                    source_id=item.source_id,
                    kind=content.kind,
                    meaning_key=content.meaning_key,
                    recall_expression=content.recall_expression,
                    text=content.text,
                    purpose=content.purpose,
                    reference_audio_id=content.reference_audio_id,
                ),
            )
        lessons = []
        for lesson in transaction.scalars(
            select(LessonRow).where(LessonRow.namespace == namespace),
        ):
            members = transaction.scalars(
                select(Membership)
                .where(Membership.lesson_id == lesson.id)
                .order_by(Membership.position),
            ).all()
            lessons.append(
                LessonInput(
                    source_id=lesson.source_id,
                    title=lesson.title,
                    target_source_ids=[target_sources[m.target_id] for m in members],
                ),
            )
        if not phrases:
            raise click.ClickException("No source targets in this namespace")
        dialogues = latest_dialogues(transaction, namespace)
        package = SourcePackage.model_validate(
            {
                "namespace": namespace,
                "language": language,
                "native_language": native_language,
                "source_revision": "preparation-export-1",
                "phrases": phrases,
                "lessons": lessons,
                "dialogues": dialogues,
            },
        )
        output.write_text(package.model_dump_json(indent=2) + "\n")

@click.command("export-vault")
@click.argument("vault_root", type=click.Path(exists=True, file_okay=False, path_type=Path))
@click.argument("manifest_path", type=click.Path(exists=True, path_type=Path))
@click.option("--output", type=click.Path(path_type=Path))
@click.option("--upload-assets", is_flag=True)
@click.pass_obj
def vault_export(
    config: Config,
    vault_root: Path,
    manifest_path: Path,
    output: Path | None,
    upload_assets: bool,
):
    """Dry-run selected stable blocks by default; write export only outside the vault."""
    resolved_vault = vault_root.resolve()
    if within_vault(config.data_dir, resolved_vault):
        raise click.ClickException("Runtime data directory must stay outside the read-only vault")
    if output and within_vault(output, resolved_vault):
        raise click.ClickException("Export output must stay outside the read-only vault")
    with runtime(config) as (jobs, media):
        manifest = VaultManifest.model_validate_json(manifest_path.read_text())
        package, report = export_vault(vault_root, manifest, media if upload_assets else None)
        if package is not None:
            with jobs.db.read() as transaction:
                report.update(source_changes(transaction, package))
        click.echo(json.dumps(report, indent=2, ensure_ascii=False))
        if package is None or report.get("identity_errors"):
            raise click.ClickException("Vault export rejected")
        if output:
            if report["assets_not_uploaded"]:
                raise click.ClickException(
                    "Use --upload-assets before writing an importable audio-bound export",
                )
            output.write_text(package.model_dump_json(indent=2) + "\n")

@phrase_gym.command("import-package")
@click.argument("package_path", type=click.Path(exists=True, dir_okay=False, path_type=Path))
@click.option(
    "--stage",
    is_flag=True,
    help="Persist the validated package for explicit publication.",
)
@click.pass_obj
def import_package(config: Config, package_path: Path, stage: bool) -> None:
    """Dry-run a manual JSON package by default; no automatic publication."""
    database = Database(config.data_dir)
    try:
        database.migrate()
        media = MediaService(database, config.data_dir)
        content = ImportService(database, PackageValidator(media.available))
        package = ManualPackage.model_validate_json(package_path.read_text(encoding="utf-8"))
        report = content.stage(package) if stage else content.validate(package)
        click.echo(report.model_dump_json(indent=2))
        if report.errors:
            raise click.ClickException("Package rejected; resolve the reported errors")
    except (ValidationError, DomainError) as error:
        raise click.ClickException(str(error)) from error
    finally:
        database.close()

@phrase_gym.command("upload-reference")
@click.argument("audio_path", type=click.Path(exists=True, dir_okay=False, path_type=Path))
@click.argument("metadata_path", type=click.Path(exists=True, dir_okay=False, path_type=Path))
@click.option(
    "--mime-type",
    type=click.Choice(["audio/wav", "audio/mpeg", "audio/ogg", "audio/mp4"]),
    required=True,
)
@click.pass_obj
def upload_reference(
    config: Config,
    audio_path: Path,
    metadata_path: Path,
    mime_type: str,
) -> None:
    """Preserve an original clip and exact text/language/provenance; returns its asset ID."""
    database = Database(config.data_dir)
    try:
        database.migrate()
        metadata = MediaMetadata.model_validate_json(metadata_path.read_text(encoding="utf-8"))
        upload = MediaUpload.model_validate(
            {
                "metadata": metadata,
                "mime_type": mime_type,
                "audio_base64": base64.b64encode(audio_path.read_bytes()).decode(),
            },
        )
        click.echo(
            MediaService(database, config.data_dir).upload(upload).model_dump_json(indent=2),
        )
    except (ValidationError, DomainError) as error:
        raise click.ClickException(str(error)) from error
    finally:
        database.close()
