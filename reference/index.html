class PhrasePackage(StrictModel):
    format: Literal["phrase-gym.package"]
    version: Literal[1]
    package_id: str = Field(pattern=r"^[A-Za-z0-9][A-Za-z0-9._-]{0,79}$")
    source_id: str = Field(pattern=r"^[A-Za-z0-9][A-Za-z0-9._-]{0,79}$")
    source_label: str = Field(min_length=1, max_length=200)
    source_link: HttpUrl | None = None
    learning_language: Literal["en"]
    card_language: Literal["en"]
    help_language: Literal["en"]
    cards: list[ImportCard] = Field(min_length=1, max_length=20)

    @field_validator("cards")
    @classmethod
    def item_ids_are_unique(cls, cards: list[ImportCard]) -> list[ImportCard]:
        ids = [card.item_id for card in cards]
        if len(ids) != len(set(ids)):
            raise ValueError("card item_id values must be unique within a package")
        return cards

def create_app() -> FastAPI:
    remote_origin = os.environ.get("PHRASE_GYM_TAILSCALE_ORIGIN", "")
    data_dir = Path(os.environ.get("PHRASE_GYM_DATA_DIR", ".data")).resolve()
    database = Database(data_dir)
    importer = Importer(database, configured_speech())
    app = FastAPI(title="Phrase Gym", version="1.0.0")
    app.state.database = database
    app.state.importer = importer
    app.include_router(import_routes(importer))
    app.include_router(recording_routes(database))
    app.include_router(study_routes(StudyService(database)))
    app.add_middleware(LocalBoundary, tailscale_origin=remote_origin)
    app.add_middleware(
        TrustedHostMiddleware,
        allowed_hosts=trusted_hosts(remote_origin),
    )

    @app.get("/api/health")
    def health() -> dict[str, str]:
        return {"status": "ok"}

    static = Path(__file__).parent / "static"
    if static.is_dir():
        app.mount("/", StaticFiles(directory=static, html=True), name="client")
    return app
